cmake_minimum_required(VERSION 3.21)
project(AtlasVoiceWorker LANGUAGES C CXX)
set(WHISPER_SOURCE_DIR "" CACHE PATH "Path to a reviewed whisper.cpp checkout")
if(NOT EXISTS "${WHISPER_SOURCE_DIR}/include/whisper.h")
    message(FATAL_ERROR "Set WHISPER_SOURCE_DIR to a local whisper.cpp checkout.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper EXCLUDE_FROM_ALL)
add_library(atlas_assistant_transcribe STATIC Transcribe.cpp)
target_include_directories(atlas_assistant_transcribe PUBLIC ../../../src)
target_compile_features(atlas_assistant_transcribe PUBLIC cxx_std_17)
target_link_libraries(atlas_assistant_transcribe PUBLIC whisper)
if(WIN32)
    add_executable(atlas-voice-worker worker.cpp)
    target_link_libraries(atlas-voice-worker PRIVATE atlas_assistant_transcribe)
endif()
option(ATLAS_ASSISTANT_SMOKE "Build the real-model local transcription smoke check" OFF)
if(ATLAS_ASSISTANT_SMOKE)
    add_executable(atlas-voice-smoke smoke.cpp)
    target_link_libraries(atlas-voice-smoke PRIVATE atlas_assistant_transcribe)
endif()
